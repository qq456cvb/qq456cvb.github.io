Sorry, this page does not exist. You can go to the [home page](/) or browse the [publications](/publications/).

<script>
  /* Project and paper pages live at /projects/x and /publications/x, not /projects/x/. When a URL with a trailing slash
     is not found, go to the same URL without it, but only if that address is a page (200), not a folder redirect or
     another 404, so this can never loop. */
  (function () {
    var p = location.pathname;
    if (p.length < 2 || p.charAt(p.length - 1) !== '/' || !window.fetch) return;
    var target = p.replace(/\/+$/, '');
    fetch(target, { method: 'HEAD', redirect: 'manual' }).then(function (r) {
      if (r.ok) location.replace(target + location.search + location.hash);
    }).catch(function () {});
  })();
</script>
